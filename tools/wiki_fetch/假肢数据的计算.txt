def diff(tup:tuple):
    return tup[1] - tup[0]


def cal1(tup:tuple, grade):
    return tup[0] + diff(tup) * grade / 100


def cal2(tup:tuple, grade):
    return -100 + 100 * tup[0] + diff(tup) * grade




def f(x, tup):
    if x <= 5:
        return 0
    elif 5 < x <= 40:
        return (x - 5) * (100 - tup[0] * 100) / (40 - 5)
    else:
        return (100 - tup[0] * 100) + (tup[1] - 1) * 20 / 11 * (x - 40)




def cal3(tup: tuple, grade):
    result = -100 + 100 * tup[0]


    if tup[1] < tup[0]:
        raise Exception("tup[1] 必须大于等于 tup[0]")


    if tup[0] >= 1 or tup[1] < 1:
        result += diff(tup) * grade
    else:
        result += min(f(grade, tup), diff(tup) * 100)


    return result


def rounding(x):  # 求最接近 x 的 5 的倍数。由于浮点数精度问题，与游戏实际结果可能有±5的误差，以游戏内为准。
    lower = int(x) // 5 * 5
    upper = lower + 5
    if x - lower <= upper - x:
        return lower
    else:
        return upper




def trim(s:str):
    return s[:-4] + s[-4:].rstrip('0')


def format_data(n):
    if n == 0:
        return ""
    else:
        return ("+" if n > 0 else "") + trim(f"{n:.6f}")
